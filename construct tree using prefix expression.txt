
# Expression Tree using Prefix Expression

# Create Node
class Node:
    def __init__(self, data):
        self.data = data
        self.left = None
        self.right = None


# Check operator
def is_operator(ch):
    return ch in ['+', '-', '*', '/']


# Construct Expression Tree
def construct_tree(prefix):
    stack = []

    for ch in reversed(prefix.split()):

        if not is_operator(ch):
            stack.append(Node(ch))

        else:
            node = Node(ch)

            node.left = stack.pop()
            node.right = stack.pop()

            stack.append(node)

    return stack.pop()


# Non-recursive Postorder Traversal
def postorder(root):
    if root is None:
        return

    stack1 = [root]
    stack2 = []

    while stack1:
        node = stack1.pop()
        stack2.append(node)

        if node.left:
            stack1.append(node.left)

        if node.right:
            stack1.append(node.right)

    while stack2:
        print(stack2.pop().data, end=" ")


# Delete entire tree
def delete_tree(root):
    if root is None:
        return None

    stack = [root]
    nodes = []

    while stack:
        node = stack.pop()
        nodes.append(node)

        if node.left:
            stack.append(node.left)

        if node.right:
            stack.append(node.right)

    for node in reversed(nodes):
        node.left = None
        node.right = None

    return None


# Main Program
prefix = "+ - A * B C D"

root = construct_tree(prefix)

print("Postorder Traversal:")
postorder(root)

root = delete_tree(root)

print("\nTree deleted successfully.")
