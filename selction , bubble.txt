salaries = [21, 34, 55, 76, 89, 95, 30]

selection = salaries.copy()
n = len(salaries)

for i in range(n):
  min_index = i

  for j in range(i + 1, n):
    if selection[j] < selection[min_index]:
      min_index = j

    selection[i], selection[min_index] = selection[min_index], selection[i]

print("Selection sort:", selection)
print("Top 5 highest salaries:", selection[-5:][::-1])

bubble = salaries.copy()
n = len(salaries)

for i in range(n):
  for j in range(0, n - i - 1):
    if bubble[j] > bubble[j + 1]:
      bubble[j], bubble[j + 1] = bubble[j + 1], bubble[j]

print("Bubble sort:", bubble)
print("Top 5 highest salaries:", bubble[-5:][::-1])
