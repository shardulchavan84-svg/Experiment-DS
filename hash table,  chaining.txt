
size = 10
table = [[] for i in range(size)]

# Hash function
def hash_function(key):
    return key % size

# Insert
def insert(key):
    index = hash_function(key)

    if key not in table[index]:
        table[index].append(key)

    print("Inserted:", key)

# Search
def search(key):
    index = hash_function(key)

    if key in table[index]:
        print("Element found at index", index)
    else:
        print("Element not found")

# Display
def display():
    for i in range(size):
        print(i, ":", table[i])

# Main program
while True:
    print("\n1. Insert")
    print("2. Search")
    print("3. Display")
    print("4. Exit")

    choice = int(input("Enter choice: "))

    if choice == 1:
        key = int(input("Enter key: "))
        insert(key)

    elif choice == 2:
        key = int(input("Enter key to search: "))
        search(key)

    elif choice == 3:
        display()

    elif choice == 4:
        break

    else:
        print("Invalid choice")
